#!/usr/bin/env python3

import argparse
import hashlib
import json
import os


def calculate_hash(path, algorithm):
    hasher = hashlib.new(algorithm)

    with open(path, "rb") as file:
        for block in iter(
            lambda: file.read(1024 * 1024),
            b""
        ):
            hasher.update(block)

    return hasher.hexdigest()


def main():
    parser = argparse.ArgumentParser(
        description="FileGuard - análisis seguro de archivos"
    )

    parser.add_argument("file")
    parser.add_argument(
        "-o",
        "--output",
        default="fileguard_report.json"
    )

    args = parser.parse_args()

    if not os.path.isfile(args.file):
        raise SystemExit("[!] Archivo no encontrado")

    stat = os.stat(args.file)

    report = {
        "tool": "FileGuard",
        "file": os.path.abspath(args.file),
        "size": stat.st_size,
        "extension": os.path.splitext(args.file)[1],
        "permissions": oct(stat.st_mode & 0o777),
        "hashes": {
            "md5": calculate_hash(args.file, "md5"),
            "sha1": calculate_hash(args.file, "sha1"),
            "sha256": calculate_hash(args.file, "sha256"),
        },
    }

    with open(
        args.output,
        "w",
        encoding="utf-8"
    ) as f:
        json.dump(
            report,
            f,
            indent=2,
            ensure_ascii=False
        )

    print(f"[+] Archivo: {report['file']}")
    print(f"[+] SHA256: {report['hashes']['sha256']}")
    print(f"[+] Reporte: {args.output}")
